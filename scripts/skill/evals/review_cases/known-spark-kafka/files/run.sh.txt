#!/usr/bin/env bash
set -euo pipefail
py/bin/odctl up spark-lite kafka-lite
# odctl up returns before Kafka answers, because the kafka service has no health check, so
# wait for the broker: 30 tries 2 seconds apart allow about a minute for it to start.
for try in $(seq 1 30); do
  if docker exec kafka /opt/kafka/bin/kafka-topics.sh --bootstrap-server broker-1:19092 --list >/dev/null 2>&1; then
    break
  fi
  if [ "$try" -eq 30 ]; then
    echo "Kafka did not answer within a minute" >&2
    exit 1
  fi
  sleep 2
done
docker cp read_orders.py spark-master:/tmp/read_orders.py
docker exec spark-master /opt/spark/bin/spark-submit \
  --packages org.apache.spark:spark-sql-kafka-0-10_2.13:4.1.1 /tmp/read_orders.py
